// Fixed Size Sliding Window Template

// 1. Build first window of size k
int window = 0;

for(int i = 0; i < k; i++) {
    window += nums[i];
}

// 2. Process first window
int answer = window;

// 3. Slide the window
for(int i = k; i < nums.size(); i++) {

    // Add new element entering window
    window += nums[i];

    // Remove element leaving window
    window -= nums[i - k];

    // Update answer
    answer = max(answer, window);
    // or min(answer, window)
}

// 4. Return answer
return answer;
